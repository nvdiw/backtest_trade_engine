python optimize.py --auto --resume -w 16

python optimize.py --mode smart --profile exit --base-source best --base-params outputs/optimize/best_params.json --tests 5000 -w 8 --output-dir outputs/optimize/refine_exit

python optimize.py --mode smart --profile signal --tests 10000 -w 8 --start 2023-01-01 --end 2025-01-01 --validation-start 2025-01-01 --validation-end 2026-01-01 --min-trades 50 --max-drawdown 35 --output-dir outputs/optimize/validated
